import ViolaSays from '@components/ViolaSays.astro'
import { Steps } from '@astrojs/starlight/components';
import { FileTree } from '@astrojs/starlight/components';

<Steps>

1. Click on the plus icon next to `assays` to add a new assay.
2. Enter an identifier for the `New Assay`.

    <ViolaSays>
        SugarsPhotometer
    </ViolaSays>

3. Click `Create Assay`.
4. ARCitISA adds the assay `SugarsPhotometer` including the folders `dataset` and `protocols` as well as a `README.md` and the `isa.assay.xlsx` workbook. You can display the file structure created by ARCitISA by clicking on `SugarsPhotometer` as shown below.

    <FileTree>
    - assays
        - **SugarsPhotometer**
            - dataset
                - ...
            - protocols
                - ...
            - isa.assay.xlsx
            - README.md
        - ...
    </FileTree>


5. Click on the assay's name (`SugarsPhotometer`) in the file tree to open the folder
6. Click on the `isa.assay.xlsx` workbook to edit the assay metadata in the right panel.
6. Here you can add metadata about your assay.

    ![](@images/arcitisa/assay.png)

    <ViolaSays>
        Ok, so my **measurement type** is "sugar measurement". 
        And to specify the device I used for my measurement, I add "photometry" and "Infinite M200 plate reader (Tecan)" as the **technology type** and **technology platform**, respectively.
    </ViolaSays>

</Steps>
