import ViolaSays from '@components/ViolaSays.astro'

import Signin from '@images/arcitisa/signin.png'

<Steps>
    
1. Click Open the DataHUB Authentication menu in the top-right.
2. Use the pre-selected `PLANTdataHUB (official)`

    <img src={Signin.src} width="50%" />

3. Click <btn> Generate Personal Access Token </btn>. This automatically opens the DataHUB in your browser and guides you to create a personal access token.
4. Scroll down and click <btn> Generate token </btn>.
5. Copy the token on top and paste it into the ARCitISA field "Personal Access Token"
6. Click <btn> Sign In </btn>

    <ViolaSays>
    Great, now I should be able to work with the DataHUB.
    </ViolaSays>

</Steps>
